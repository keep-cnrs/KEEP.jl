\section{Explicit dynamics} \label{sec:explicit-ddq}

This appendix gives the system~\eqref{eq:eom_regular} and its solution.
We use the notation of Section~\ref{sec:model-geometry}: \(p = \vu r\vdot\vb a\), \(r = p + R\) with \(R = \sqrt{p^2+\ell^2-a^2}\), \(\ell\hat{\boldsymbol{\ell}} = r\vu r - a\vu a\), the lever arm \(\lambda\) of~\eqref{eq:lever-arm}, and
\begin{equation}
    v = \norm{\vu r'} > 0, \qquad r_\alpha = \pdv{r}{\alpha}, \qquad r_\tau = \pdv{r}{\tau},
\end{equation}
with \(\nabla^2 r\) the Hessian of \(r(\alpha,\tau)\).
The force \(\vb F = \vb F_a + \vb F_g\) depends on \((\vq, \dq)\) only.

% \paragraph{Three identities.}
\begin{enumerate}
    \item \(\hat{\boldsymbol{\ell}}\vdot\vu r = R/\ell\), since \(\ell\hat{\boldsymbol{\ell}}\vdot\vu r = r - p = R\).
    \item \(r_\alpha = (a\ell/R)\,\lambda\): indeed \(\pdv*{p}{\alpha} = a\,\vu r\vdot(\vu z\times\vu a)\), \(\vu r\vdot(\vu z\times\vu a) = \ell\lambda/r\) by~\eqref{eq:lever-arm}, and \(\dv*{r}{p} = r/R\). So \(r_\alpha\) vanishes on the fold curves at the same rate as \(\lambda\).
    \item \(r_\tau = (r/R)\,a\,\vu a\vdot\vu r'\) and, as \(\vu r\vdot\vu r' = 0\), \(\hat{\boldsymbol{\ell}}\vdot\vu r' = -(a/\ell)\,\vu a\vdot\vu r' = -R\,r_\tau/(r\ell)\).
\end{enumerate}

% \paragraph{The three scalar equations.}
With \(\vb r = r\vu r(\tau)\), \(\vu r\vdot\vu r' = 0\) and \(\vu r\vdot\vu r'' = -v^2\),
\begin{align}
    \ddot{\vb r}\vdot\vu r &= r_\alpha\ddot\alpha + r_\tau\ddot\tau + \dq\vdot\nabla^2 r\,\dq - r v^2\dot\tau^2, \\
    \ddot{\vb r}\vdot\vu r' &= r v^2\ddot\tau + 2\dot r v^2\dot\tau + r\,(\vu r'\vdot\vu r'')\,\dot\tau^2.
\end{align}
The projections of~\eqref{eq:newton_kite} onto \(\vu r\) and \(\vu r'\), with \(\vb F_\ell = -F_\ell\hat{\boldsymbol{\ell}}\) and the first identity, and the equation~\eqref{eq:newton_arm} of the arm read
\begin{align}
    m\,(r_\alpha\ddot\alpha + r_\tau\ddot\tau) + \frac{R}{\ell}F_\ell &= c_1, &
    c_1 &= -m\,(\dq\vdot\nabla^2 r\,\dq - r v^2\dot\tau^2) + \vu r\vdot\vb F, \label{eq:K1} \\
    m r v^2\ddot\tau + (\hat{\boldsymbol{\ell}}\vdot\vu r')\,F_\ell &= c_2, &
    c_2 &= -m\,(2\dot r v^2\dot\tau + r\,(\vu r'\vdot\vu r'')\,\dot\tau^2) + \vu r'\vdot\vb F, \label{eq:K2} \\
    I\ddot\alpha - a\lambda F_\ell &= T(\dot\alpha). \label{eq:A}
\end{align}

% \paragraph{Regular form.}
Substituting~\eqref{eq:ddalpha}, that is~\eqref{eq:A} solved for \(\ddot\alpha\), into~\eqref{eq:K1}, and using the second identity, \(m r_\alpha a\lambda/I = m a^2\ell\lambda^2/(IR)\), gives~\eqref{eq:eom_regular} with
\begin{equation}
    N(\vq) = \mqty(m r_\tau & \dfrac{R}{\ell} + \dfrac{m a^2\ell\lambda^2}{IR} \\ m r v^2 & \hat{\boldsymbol{\ell}}\vdot\vu r'),
    \qquad
    \vb h(\vq,\dq) = \mqty(c_1 - m r_\alpha T(\dot\alpha)/I \\ c_2).
\end{equation}
By the third identity,
\begin{equation} \label{eq:detN}
    \det N = -\frac{m R\,r_\tau^2}{r\ell} - \frac{m r v^2 R}{\ell} - \frac{m^2 a^2\ell r v^2\lambda^2}{IR} \leq -\frac{m r v^2 R}{\ell} < 0.
\end{equation}
As \(r\geq\ell-a\), \(R\geq\sqrt{\ell^2-a^2}\) and \(v\geq v_*\) (Appendix~\ref{app:curv}),
\begin{equation}
    \abs{\det N} \geq \frac{m\,(\ell-a)\,v_*^2\sqrt{\ell^2-a^2}}{\ell} > 0
\end{equation}
on the whole torus of configurations.
By Cramer's rule, with \(h_1 = c_1 - m r_\alpha T(\dot\alpha)/I\),
\begin{align}
    \ddot\tau &= \frac{1}{\det N}\qty[h_1\,(\hat{\boldsymbol{\ell}}\vdot\vu r') - c_2\qty(\frac{R}{\ell} + \frac{m a^2\ell\lambda^2}{IR})], \\
    F_\ell &= \frac{1}{\det N}\qty[m r_\tau c_2 - m r v^2 h_1], \\
    \ddot\alpha &= \frac{a\lambda F_\ell + T(\dot\alpha)}{I}.
\end{align}
These expressions are analytic wherever the force \(\vb F\) is, since the kinematic quantities \(r\), \(R\), \(v\) are positive and analytic on the whole torus.
Off the fold curves (\(\lambda\neq0\)), they are equivalent to the elimination of the tension through~\eqref{eq:A}, and define the same flow; on the fold curves, the arm is free of the lines, \(I\ddot\alpha = T(\dot\alpha)\), and the tension is fixed by the balance~\eqref{eq:K1}--\eqref{eq:K2} of the kite.
They are implemented as such in the code (\texttt{PointMass4.dynamics}), at the cost of one \(2\times2\) solve per evaluation, and the tension along a trajectory is given by the same solve.
